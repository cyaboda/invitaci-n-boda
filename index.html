function doGet(e) { const hoja = SpreadsheetApp .openById("AQUI_VA_EL_ID_DE_TU_GOOGLE_SHEET") .getSheetByName("INVITADOS"); let resultado; if (!hoja) { resultado = { ok: false, mensaje: "No se encontró la hoja INVITADOS." }; } else { const codigo = e.parameter.codigo; if (!codigo) { resultado = { ok: false, mensaje: "No se recibió ningún código." }; } else { const datos = hoja.getDataRange().getValues(); resultado = { ok: false, mensaje: "Invitación no encontrada." }; for (let i = 1; i < datos.length; i++) { if ( String(datos[i][0]).trim().toUpperCase() === String(codigo).trim().toUpperCase() ) { resultado = { ok: true, codigo: datos[i][0], invitado: datos[i][1], cupos: datos[i][2], estado: datos[i][3] || "PENDIENTE", asisten: datos[i][4] || "", fechaRespuesta: datos[i][5] || "" }; break; } } } } const callback = e.parameter.callback; if (callback) { return ContentService .createTextOutput( callback + "(" + JSON.stringify(resultado) + ");" ) .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); }
